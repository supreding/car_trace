#   .venv-reid/bin/pip install "torch==2.12.1+cu126" "torchvision==0.27.1+cu126" \
#       --find-links https://mirrors.aliyun.com/pytorch-wheels/cu126/ -i https://mirrors.aliyun.com/pypi/simple/
#   .venv-reid/bin/pip install paddlepaddle-gpu==3.3.1 -i https://www.paddlepaddle.org.cn/packages/stable/cu126/
#   .venv-reid/bin/pip install -r requirements-reid.txt -i https://mirrors.aliyun.com/pypi/simple/
#
# Windows 离线安装：先在有网机器 pip download -r requirements-reid.txt -d _wheels，
